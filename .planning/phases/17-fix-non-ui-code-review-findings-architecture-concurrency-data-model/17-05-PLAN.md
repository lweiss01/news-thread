  <action>
    Create an instrumented test using `MigrationTestHelper`:

    ```kotlin
    @RunWith(AndroidJUnit4::class)
    class MigrationTest {
        @get:Rule
        val helper = MigrationTestHelper(
            InstrumentationRegistry.getInstrumentation(),
            AppDatabase::class.java
        )

        @Test
        fun migrate_publishedAt_string_to_long() {
            // Create DB at version N with String publishedAt
            val db = helper.createDatabase(TEST_DB, VERSION_N)
            db.execSQL("INSERT INTO cached_articles (..., publishedAt, ...) VALUES (..., '1709000000000', ...)")
            db.execSQL("INSERT INTO cached_articles (..., publishedAt, ...) VALUES (..., 'not_a_number', ...)")
            db.close()

            // Migrate to version N+1
            val migratedDb = helper.runMigrationsAndValidate(TEST_DB, VERSION_N1, true, MIGRATION_N_N1)

            // Verify: numeric string → preserved as Long
            val cursor = migratedDb.query("SELECT publishedAt FROM cached_articles")
            cursor.moveToFirst()
            assertEquals(1709000000000L, cursor.getLong(0))

            // Verify: non-numeric string → fallback to 0
            cursor.moveToNext()
            assertEquals(0L, cursor.getLong(0))
        }
    }
    ```

    Check `app/build.gradle.kts` for androidTest dependencies (`room-testing`). Add if missing:
    ```kotlin

    ```
